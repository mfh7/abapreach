Usage: abapreach deploy [OPTIONS] --conn <CONN> <KIND> <NAME> <FILE>

Arguments:
  <KIND>
          Possible values:
          - PROG
          - FUNC
          - CLAS
          - INTF
          - TRAN: A report transaction; it has no ADT path and runs over the RFC EXT path only

  <NAME>
          

  <FILE>
          

Options:
      --simulate
          

      --base <BASE>
          

      --tr <TR>
          

      --backend <BACKEND>
          [possible values: adt, rfc]

      --include <INCLUDE>
          [possible values: CCIMP, CCDEF, CCMAC, CCAU]

      --create-include
          

      --fugr <FUGR>
          

      --main <MAIN>
          The main program of a `PROG` include, as `INCLUDE <NAME>.` names it

      --conn <CONN>
          

  -h, --help
          Print help (see a summary with '-h')
