Usage: abapreach deploy [OPTIONS] --conn <CONN> <KIND> <NAME> <FILE>

Arguments:
  <KIND>  [possible values: PROG, FUNC, CLAS, INTF, DTEL, DOMA, TABL, TTYP, TRAN]
  <NAME>  
  <FILE>  

Options:
      --simulate           For DTEL, DOMA, TABL, TTYP and TRAN a simulate still creates and deletes a temporary class in $TMP
      --base <BASE>        
      --tr <TR>            
      --backend <BACKEND>  [possible values: adt, rfc]
      --include <INCLUDE>  [possible values: CCIMP, CCDEF, CCMAC, CCAU]
      --create-include     
      --fugr <FUGR>        
      --main <MAIN>        The main program of a `PROG` include, as `INCLUDE <NAME>.` names it
      --conn <CONN>        
  -h, --help               Print help
